// Load initial keyframe first for instant render

  ? await createImageBitmap(firstBlob)

// Stream the remaining frames in non-blocking batches
const BATCH_SIZE = 8;
for (let b = 0; b < TOTAL_FRAMES; b += BATCH_SIZE) {
  const batchPromises = [];
  // … fetch → blob → createImageBitmap for each frame in the batch
  await Promise.all(batchPromises);
}
